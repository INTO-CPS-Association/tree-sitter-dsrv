out speed: Int = match (s) {
  Stopped -> 0,
  Moving(v) if v > 10 -> 10,
  Moving(v) -> v,
  Turning((a, _)) | Turning((_, a)) -> a,
}
out band: Str = match (n,) {
  -5..0 -> "negative",
  0 -> "zero",
  1..=9 -> "small",
  whole @ _ -> "large",
}
out shape: Bool = match (l) {
  [] -> true,
  [x, y] -> x == y,
  _ -> false
}
out field: Int = match (p) {
  {a: 1, ..} -> 1,
  {a, b: "x"} -> a,
  {..} -> 0,
}
out stopped: Bool = matches(s, Stopped | Moving(0))
out fast: Bool = matches(s, Moving(v) if v > 5,)
out tagged: State = State::Moving(n)
